{{- /* The title block: on a real drawing this is the ruled box in the corner that states what the sheet is, who drew it, when, and at which revision. Here it carries the post's metadata as an actual bordered table instead of the usual grey caption line, which is the whole thesis of the theme. Every row is a label/value pair, so the block reads correctly as a definition list to a screen reader and stays a table to the eye. Rows that have no value are omitted rather than left blank — an empty field on a drawing means "not applicable", and a missing one means "not recorded". Revision is .Lastmod, and only appears when it genuinely differs from the publication date; otherwise the sheet is at its first issue. */ -}} {{- $rows := slice }} {{- if not .Date.IsZero }} {{- $rows = $rows | append (dict "label" (i18n "tb_issued") "value" (printf `%s` (.Date.Format "2006-01-02T15:04:05Z07:00") (.Date) (.Date | time.Format (site.Params.DateFormat | default ":date_long")))) }} {{- end }} {{- if and (not .Lastmod.IsZero) (not .Date.IsZero) (gt (.Lastmod.Sub .Date).Hours 24.0) }} {{- $rows = $rows | append (dict "label" (i18n "tb_revised") "value" (printf `%s` (.Lastmod.Format "2006-01-02T15:04:05Z07:00") (.Lastmod | time.Format (site.Params.DateFormat | default ":date_long")))) }} {{- end }} {{- $scale := slice }} {{- if and (.Param "ShowReadingTime") (gt .ReadingTime 0) }} {{- $scale = $scale | append (i18n "read_time" .ReadingTime) }} {{- end }} {{- if and (.Param "ShowWordCount") (gt .WordCount 0) }} {{- $scale = $scale | append (i18n "words" .WordCount) }} {{- end }} {{- with $scale }} {{- $rows = $rows | append (dict "label" (i18n "tb_extent") "value" (delimit . " · ")) }} {{- end }} {{- with partial "author.html" . }} {{- $rows = $rows | append (dict "label" (i18n "tb_drawn_by") "value" .) }} {{- end }} {{- /* [taxonomies] is site config, not params, so the old site.Language.Params.Taxonomies.tag lookup here was always nil and the default always won.
