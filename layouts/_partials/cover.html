{{- /* The post's cover image. Resolved from the page bundle first, then from the theme's or site's assets, so a bundled cover.png needs no front matter at all. Covers are eager on the post itself (it is the largest contentful paint) and lazy in lists, where several are below the fold. Usage: {{ partial "cover.html" (dict "page" $p "isSingle" true) }} */ -}} {{- $p := .page }} {{- $single := .isSingle }} {{- $cover := $p.Params.cover }} {{- $hidden := false }} {{- if reflect.IsMap $cover }} {{- $hidden = or $cover.hidden (and $single $cover.hiddenInSingle) (and (not $single) $cover.hiddenInList) }} {{- end }} {{- if not $hidden }} {{- $name := "" }} {{- if reflect.IsMap $cover }}{{ $name = $cover.image }}{{ else }}{{ $name = $cover }}{{ end }} {{- $img := "" }} {{- if $name }} {{- with $p.Resources.GetMatch $name }}{{ $img = . }}{{ end }} {{- if not $img }}{{ with resources.Get $name }}{{ $img = . }}{{ end }}{{ end }} {{- else }} {{- with $p.Resources.GetMatch "{cover,cover.*,*-cover.*}" }}{{ $img = . }}{{ end }} {{- end }} {{- if and $img (ne $img.ResourceType "image") }}{{ $img = "" }}{{ end }} {{- if $img }} {{- $t := partialCached "tokens.html" $p "tokens" }} {{- $set := partial "image-set.html" (dict "resource" $img "measure" $t.measure) }} {{- $alt := "" }} {{- if reflect.IsMap $cover }}{{ $alt = $cover.alt | default $cover.caption }}{{ end }} {{- $caption := "" }} {{- if reflect.IsMap $cover }}{{ $caption = $cover.caption }}{{ end }} {{- end }} {{- end }}
