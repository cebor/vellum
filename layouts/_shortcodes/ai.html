{{- /* Marks a passage as AI-assisted, the way a drawing marks a changed region: a change bar down its edge, the sheet's AI mark flagged on the bar, and a number resolving to a row in the revision note at the foot of the sheet. The flag links to its row, because that is what a drawing reference is for — it tells you where to look. {{< ai >}}A whole paragraph.{{< /ai >}} … a sentence with {{< ai display="inline" note="translated" >}}this run {{< /ai >}} inside it. `unsafe` is off, so the inner text goes through RenderString exactly as the collapse shortcode does, and stays ordinary Markdown in the source. */ -}} {{- $display := .Get "display" | default "block" }} {{- if not (in (slice "block" "inline") $display) }} {{- errorf "ai shortcode in %s: display must be %q or %q, not %q" .Page.Path "block" "inline" $display }} {{- end }} {{- $ai := partialCached "ai.html" .Page .Page.RelPermalink }} {{- $note := .Get "note" | default $ai.note }} {{- $model := .Get "model" | default $ai.model }} {{- /* Numbering is derived from .Ordinal rather than kept as a running count, so it is idempotent: Page.Store outlives a page, and a second render of the same content for another output format would otherwise carry the counter on and number the same passage twice. The key is zero-padded so that the map's own key order is already document order, and the revision note needs no sort. */ -}} {{- $key := printf "%03d" .Ordinal }} {{- $marks := .Page.Store.Get "aiMarks" | default dict }} {{- $marks = merge $marks (dict $key (dict "note" $note "model" $model)) }} {{- .Page.Store.Set "aiMarks" $marks }} {{- $n := 0 }} {{- range $k, $_ := $marks }}{{ if le $k $key }}{{ $n = add $n 1 }}{{ end }}{{ end }} {{- $label := i18n "ai_flag" (dict "Number" $n) }} {{- if eq $display "inline" -}} {{- /* In running prose there is no change bar for a flag to sit on, so the mark is a reference rather than a flag: the dashed line under the words, and the number raised after them. A glyph set mid-sentence would stop the line dead, and the dashed underline already says which words are marked. */ -}} {{ .Inner | .Page.RenderString (dict "display" "inline") }}{{ $n }} {{- else -}} {{- /* The mark on the bar is the sheet's own AI mark — the same nib-over-plotted- line the title block stamps in its corner — so one mark says "AI" here, once for the whole sheet and once for the passage. The number rides after it as the reference it is, the same raised figure the inline case sets. */ -}} {{- $flag := printf `%s%d` $n $label (partialCached "icon.html" "ai" "ai") $n }}
