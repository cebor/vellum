The flag links to its row, because that is what a drawing reference is for — it tells you where to look. {{< ai >}}A whole paragraph.{{< /ai >}} … a sentence with {{< ai display="inline" note="translated" >}}this run {{< /ai >}} inside it. `unsafe` is off, so the inner text goes through RenderString exactly as the collapse shortcode does, and stays ordinary Markdown in the source. */ -}} {{- $display := .Get "display" | default "block" }} {{- if not (in (slice "block" "inline") $display) }} {{- errorf "ai shortcode in %s: display must be %q or %q, not %q" .Page.Path "block" "inline" $display }} {{- end }} {{- $ai := partialCached "ai.html" .Page .Page.RelPermalink }} {{- $note := .Get "note" | default $ai.note }} {{- $model := .Get "model" | default $ai.model }} {{- /* Numbering is derived from .Ordinal rather than kept as a running count, so it is idempotent: Page.Store outlives a page, and a second render of the same content for another output format would otherwise carry the counter on and number the same passage twice. The key is zero-padded so that the map's own key order is already document order, and the revision note needs no sort. */ -}} {{- $key := printf "%03d" .Ordinal }} {{- $marks := .Page.Store.Get "aiMarks" | default dict }} {{- $marks = merge $marks (dict $key (dict "note" $note "model" $model)) }} {{- .Page.Store.Set "aiMarks" $marks }} {{- $n := 0 }} {{- range $k, $_ := $marks }}{{ if le $k $key }}{{ $n = add $n 1 }}{{ end }}{{ end }} {{- $label := i18n "ai_flag" (dict "Number" $n) }} {{- if eq $display "inline" -}} {{- /* In running prose there is no change bar for a flag to sit on, so the mark is a reference rather than a flag: the dashed line under the words, and the number raised after them. A triangle set mid-sentence reads as a warning sign and stops the line dead. */ -}} {{ .Inner | .Page.RenderString (dict "display" "inline") }}{{ $n }} {{- else -}} {{- $flag := printf `%s%d` $n $label (partialCached "icon.html" "revision" "revision") $n }}
